cmake_minimum_required(VERSION 3.24)
project(SwiftPrismLoraSharing LANGUAGES C CXX)
if(NOT DEFINED PRISM_SOURCE_DIR)
  message(FATAL_ERROR "PRISM_SOURCE_DIR must name the verified, overlaid pinned source")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PRISM_SOURCE_DIR}/ggml" "${CMAKE_CURRENT_BINARY_DIR}/ggml")
add_executable(prism-lora-share-tests tests.cpp)
target_include_directories(prism-lora-share-tests PRIVATE "${PRISM_SOURCE_DIR}/src")
target_link_libraries(prism-lora-share-tests PRIVATE ggml ggml-cpu)
if(MSVC)
  target_compile_options(prism-lora-share-tests PRIVATE /W4 /WX /permissive- /sdl /fp:strict)
else()
  target_compile_options(prism-lora-share-tests PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
enable_testing()
add_test(NAME prism-lora-storage-alias COMMAND prism-lora-share-tests)
