cmake_minimum_required(VERSION 3.24)
project(SwiftBitcosPrototype LANGUAGES CXX)
option(BITCOS_DOTPROD "Require ARM dot-product instructions on the executing host" OFF)
add_library(bitcos STATIC bitcos.cpp)
target_compile_features(bitcos PUBLIC cxx_std_20)
target_include_directories(bitcos PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(bitcos PRIVATE BITCOS_DOTPROD=$<BOOL:${BITCOS_DOTPROD}>)
add_executable(bitcos-tests tests.cpp)
target_link_libraries(bitcos-tests PRIVATE bitcos)
add_executable(bitcos-bench benchmark.cpp)
target_link_libraries(bitcos-bench PRIVATE bitcos)
foreach(target bitcos bitcos-tests bitcos-bench)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /sdl /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
    if(BITCOS_DOTPROD)
      target_compile_options(${target} PRIVATE -march=armv8.2-a+dotprod)
    endif()
  endif()
endforeach()
enable_testing()
add_test(NAME bitcos-all-correctness COMMAND bitcos-tests all)
