cmake_minimum_required(VERSION 3.20)
project(bonsai_qwen_bridge LANGUAGES CXX)
if(NOT DEFINED BONSAI_TTS_SOURCE_DIR)
    message(FATAL_ERROR "An explicitly provisioned Qwen source directory is required")
endif()
add_library(bonsai-qwen-bridge STATIC bridge.cpp)
target_compile_features(bonsai-qwen-bridge PRIVATE cxx_std_17)
target_compile_definitions(bonsai-qwen-bridge PRIVATE QWEN_STATIC)
target_include_directories(bonsai-qwen-bridge PRIVATE "${BONSAI_TTS_SOURCE_DIR}/src")
if(MSVC)
    target_compile_options(bonsai-qwen-bridge PRIVATE /utf-8 /W4 /WX)
endif()
